\section{Beyond Programming Problems: MDPs}

\begin{frame}{Where dynamic programming was invented}
	Bellman's original setting: \alert{sequential decisions under uncertainty}.

	\vspace{0.5em}
	Three assumptions from Part~1 are about to break:
	\begin{enumerate}
		\item transitions are \textbf{stochastic};
		\item the horizon may be \textbf{unbounded}, so the state graph has \textbf{cycles};
		\item the state space may be too large to sweep.
	\end{enumerate}
\end{frame}

\begin{frame}{Markov decision processes}
	\begin{block}{Definition}
		$\mathcal M=(\mathcal S,\mathcal A,P,r,\gamma)$ with finite $\mathcal S,\mathcal A$; transition kernel
		$P(s'\mid s,a)$; expected reward $r(s,a)$; discount $\gamma\in[0,1)$.
	\end{block}
\end{frame}

\begin{frame}{Policies and values}
	A (stationary) policy $\pi(a\mid s)$ induces the return $G_t=\sum_{k\ge0}\gamma^kR_{t+k+1}$ and
	\[
		v_\pi(s)=\expect[\pi]{G_t\mid S_t=s},\qquad
		q_\pi(s,a)=\expect[\pi]{G_t\mid S_t=s,A_t=a}.
	\]
	Goal: $v^\star(s)=\max_\pi v_\pi(s)$ for all $s$ simultaneously.
\end{frame}

\begin{frame}{Edit distance as an MDP}
	\audience[Edit distance, revisited]{%
		\begin{enumerate}
			\item Cast the edit-distance problem as an MDP: what are $\mathcal S$, $\mathcal A$, $P$, $r$, $\gamma$?
			\item Now the diagonal action \emph{fails} with probability $\varepsilon$ and leaves you in place.
			What happens to the state graph?
		\end{enumerate}
	}
\end{frame}

\begin{frame}{Edit distance as an MDP: answer}
	\begin{enumerate}
		\item $\mathcal S=\{0..m\}\times\{0..n\}$, $\mathcal A=\{\text{del},\text{ins},\text{diag}\}$, deterministic $P$,
		$r=-\text{cost}$, $\gamma=1$, target absorbing. A policy is a path.
		\item $(i,j)$ gets a \alert{self-loop} with probability $\varepsilon$. The graph is no longer a DAG:
		$v(i,j)$ appears on both sides of its own equation.
	\end{enumerate}
\end{frame}

\begin{frame}{The Bellman expectation equation}
	Since $G_t=R_{t+1}+\gamma G_{t+1}$, the tower property and the Markov property give
	\[
		v_\pi(s)=\sum_a\pi(a\mid s)\Bigl[r(s,a)+\gamma\sum_{s'}P(s'\mid s,a)\,v_\pi(s')\Bigr].
	\]
\end{frame}

\begin{frame}{The Bellman expectation equation in matrix form}
	With $P_\pi(s,s')=\sum_a\pi(a|s)P(s'|s,a)$ and $r_\pi$ similarly:
	\[
		v_\pi=r_\pi+\gamma P_\pi v_\pi\quad\Longleftrightarrow\quad v_\pi=(I-\gamma P_\pi)^{-1}r_\pi.
	\]
	$I-\gamma P_\pi$ is invertible since the spectral radius of $\gamma P_\pi$ is $\gamma<1$.

	\seprule

	$|\mathcal S|$ \alert{simultaneous} linear equations, rather than an ordered recurrence.
\end{frame}

\begin{frame}{The Bellman optimality equation}
	\[
		v^\star(s)=\max_{a}\Bigl[r(s,a)+\gamma\sum_{s'}P(s'\mid s,a)\,v^\star(s')\Bigr]
	\]
	\[
		q^\star(s,a)=r(s,a)+\gamma\sum_{s'}P(s'\mid s,a)\max_{a'}q^\star(s',a').
	\]
	Any policy greedy with respect to $v^\star$ is optimal:
	\[
		\pi^\star(s)\in\arg\max_a\Bigl[r(s,a)+\gamma\sum_{s'}P(s'|s,a)v^\star(s')\Bigr].
	\]
\end{frame}

\begin{frame}{Back to Part 1}
	\begin{itemize}
		\item \textbf{Optimal substructure} becomes the \emph{principle of optimality}: a tail of an optimal policy is optimal from the state where the tail starts.
		\item \textbf{Reconstruction} becomes the $\arg\max$: the policy is recovered from the values.
		\item Deterministic transitions and $\gamma=1$ give back $d(v)=\min\{d(u)+w\}$.
	\end{itemize}
\end{frame}

\begin{frame}{Finite horizon: Part 1 verbatim}
	Horizon $T$, terminal reward $r_T(s)$. The state space $(t,s)$ is a \alert{DAG}.
	\begin{align*}
		V_T(s)&=r_T(s) &&\text{base case}\\
		V_t(s)&=\max_a\Bigl[r_t(s,a)+\sum_{s'}P_t(s'|s,a)\,V_{t+1}(s')\Bigr] &&\text{inductive step}
	\end{align*}
	Evaluate for $t=T-1,\dots,0$; the optimal policy is the argmax.

	\seprule

	Cost: $\Theta(T\,|\mathcal S|\,|\mathcal A|\,b)$ with branching factor $b\le|\mathcal S|$.
\end{frame}

\begin{frame}{Infinite horizon: the DAG disappears}
	With $T=\infty$, the equations for $v^\star$ are cyclic: no base case, no topological order.

	\vspace{0.5em}
	The \alert{Bellman operator}:
	\[
		(\Bell V)(s)=\max_a\Bigl[r(s,a)+\gamma\sum_{s'}P(s'|s,a)V(s')\Bigr].
	\]
\end{frame}

\begin{frame}{The Bellman operator is a contraction}
	\begin{theorem}
		$\Bell$ is a $\gamma$-contraction in $\supnorm\cdot$. Hence $v^\star$ is its unique fixed point and
		$\supnorm{\Bell^kV_0-v^\star}\le\gamma^k\supnorm{V_0-v^\star}$ for every $V_0$.
	\end{theorem}
	\begin{proof}
		Use $|\max_af(a)-\max_ag(a)|\le\max_a|f(a)-g(a)|$:
		\[
			|(\Bell V)(s)-(\Bell U)(s)|\le\gamma\max_a\sum_{s'}P(s'|s,a)\,|V(s')-U(s')|\le\gamma\supnorm{V-U}.\qedhere
		\]
	\end{proof}
\end{frame}

\begin{frame}{Value iteration}
	\[
		V_{k+1}=\Bell V_k,\qquad V_0\text{ arbitrary}.
	\]
	One sweep costs $\Theta(|\mathcal S|\,|\mathcal A|\,b)$.
\end{frame}

\begin{frame}{Value iteration: when to stop}
	A computable error certificate:
	\[
		\supnorm{V_{k+1}-v^\star}\le\tfrac{\gamma}{1-\gamma}\supnorm{V_{k+1}-V_k}.
	\]
	\begin{proof}
		$\supnorm{V_{k+1}-v^\star}\le\supnorm{V_{k+1}-\Bell V_{k+1}}+\supnorm{\Bell V_{k+1}-\Bell v^\star}
		\le\gamma\supnorm{V_k-V_{k+1}}+\gamma\supnorm{V_{k+1}-v^\star}$.
	\end{proof}
	For the greedy policy $\pi_V$: $\supnorm{v_{\pi_V}-v^\star}\le\tfrac{2}{1-\gamma}\supnorm{\Bell V-V}$.
\end{frame}

\begin{frame}{Policy iteration}
	Alternate \alert{evaluation} (solve $v_\pi=(I-\gamma P_\pi)^{-1}r_\pi$) and \alert{improvement}
	($\pi'(s)\in\arg\max_a q_\pi(s,a)$).

	\begin{theorem}[policy improvement]
		$v_{\pi'}\ge v_\pi$ pointwise, strictly somewhere unless $\pi$ is optimal.
	\end{theorem}
	\begin{proof}[Sketch]
		$\Bell_{\pi'}v_\pi=\Bell v_\pi\ge v_\pi$ and $\Bell_{\pi'}$ is monotone, so
		$v_\pi\le\Bell_{\pi'}v_\pi\le\Bell_{\pi'}^2v_\pi\le\dots\to v_{\pi'}$.
	\end{proof}
\end{frame}

\begin{frame}{Policy iteration on the example}
	Start from $(a,c)$:
	\[
		v=(10,20)\;\Rightarrow\;\text{improve to }(b,c)\;\Rightarrow\;v=(17.561,20),
	\]
	stable after \alert{two} iterations.
\end{frame}

\begin{frame}{Asynchronous dynamic programming}
	Value iteration is \emph{synchronous}. Assumptions we drop:
	\begin{enumerate}
		\item a \alert{global sweep} that touches all $|\mathcal S|$ states each iteration;
		\item a \alert{fixed schedule} and a barrier between iterations;
		\item all states are equally worth updating.
	\end{enumerate}
\end{frame}

\begin{frame}{The asynchronous update}
	At each step choose any state $s_t$ and set
	\[
		V(s_t)\leftarrow(\Bell V)(s_t),
	\]
	using whatever values of the other states are currently available.

	\seprule

	Order, frequency and staleness of reads are free.
\end{frame}

\begin{frame}{Schedules for asynchronous updates}
	\begin{itemize}
		\item \textbf{In place (Gauss--Seidel):} use updated values immediately.
		\item \textbf{Prioritised sweeping:} update the state with the largest Bellman error
		$|(\Bell V)(s)-V(s)|$ (Moore--Atkeson 1993).
		\item \textbf{Real-time DP (RTDP):} update only along trajectories of the greedy policy
		(Barto--Bradtke--Singh 1995).
		\item \textbf{Heuristic search:} A*/LAO* are RTDP-like schedules on AND/OR graphs.
	\end{itemize}
\end{frame}

\begin{frame}{Dijkstra is asynchronous value iteration}
	\begin{itemize}
		\item \textbf{Dijkstra:} schedule ``finalise the state with the smallest tentative value''.
		Each state is updated once.
		\item \textbf{Bellman--Ford:} the synchronous schedule.
	\end{itemize}
\end{frame}

\begin{frame}{Choose your schedule: answer}
	\begin{enumerate}
		\item \textbf{Three sweeps}: information travels one edge per sweep.
		\item $c,b,a$: a \alert{reverse topological order}.
	\end{enumerate}

	\seprule

	Bottom-up evaluation is the best possible asynchronous schedule on a DAG.
\end{frame}

\begin{frame}{Why asynchronous updates converge}
	\begin{theorem}[Bertsekas 1982; Bertsekas--Tsitsiklis 1989]
		Let $\Bell$ be a $\gamma$-contraction in $\supnorm\cdot$. If each coordinate is updated infinitely often, and
		stale information is eventually flushed, then $V_t\to v^\star$ for any initial $V_0$.
	\end{theorem}
\end{frame}

\begin{frame}{Proof: the box argument}
	\small
	Let $D=\supnorm{V_0-v^\star}$ and $B_k=\{V:\supnorm{V-v^\star}\le\gamma^kD\}$.
	Let $t_k$ be a time after which every coordinate has been updated at least once using only information from
	after $t_{k-1}$.

	\vspace{0.5em}
	\textbf{Claim:} $V_t\in B_k$ for $t\ge t_k$.

	\vspace{0.5em}
	\textbf{Induction:} an update of coordinate $s$ reads a vector in $B_{k-1}$, so
	\[
		|V_{t+1}(s)-v^\star(s)|=|(\Bell V)(s)-(\Bell v^\star)(s)|\le\gamma\cdot\gamma^{k-1}D,
	\]
	and further updates keep $s$ within the box.\qed
\end{frame}

\begin{frame}{From asynchronous DP to learning}
	Replace the expectation over $P$ by a \emph{sample} $(s,a,r,s')$ and take a step of size $\alpha_t$:
	\[
		Q(s,a)\leftarrow Q(s,a)+\alpha_t\Bigl[r+\gamma\max_{a'}Q(s',a')-Q(s,a)\Bigr].
	\]
	This is \alert{Q-learning} (Watkins--Dayan 1992; Tsitsiklis 1994).
	It converges when every pair $(s,a)$ is visited infinitely often and $\sum\alpha_t=\infty$, $\sum\alpha_t^2<\infty$.
\end{frame}

\end{frame}
