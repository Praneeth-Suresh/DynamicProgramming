\section{Going Faster}

\begin{frame}{Where does the time go?}
	Time $\approx\sum_v \text{indeg}(v)=|E|$. Two levers, both about the \emph{shape} of the DAG:

	\begin{enumerate}
		\item[\alert{I}] \textbf{Change the recurrence.} Turn
		``minimum over all predecessors'' into a structured query.
		\item[\alert{II}] \textbf{Change the compute order.} Add intermediate states so
		that every state has $O(1)$ in-edges.
	\end{enumerate}
\end{frame}

% ---------------------------------------------------------------- Lever I
\begin{frame}{Lever I: a problem with a quadratic transition}
	\textbf{Batched jobs.} Jobs $1,\dots,n$ are processed in order; job $l$ has size $c_l\ge0$ and $S_i=\sum_{l\le i}c_l$.
	Partition the sequence into consecutive batches. A batch of total size $s$ costs $M+s^2$.

	\[
		\mathrm{dp}(0)=0,\qquad
		\mathrm{dp}(i)=\min_{0\le j<i}\Bigl\{\,\mathrm{dp}(j)+(S_i-S_j)^2+M\,\Bigr\}.
	\]

	$n$ states, $\Theta(n)$ predecessors each: $\Theta(n^2)$.
\end{frame}

\begin{frame}{Lever I: where is the coupling?}
	\audience[Expand the square]{%
		\[
			\mathrm{dp}(i)=\min_{0\le j<i}\Bigl\{\,\mathrm{dp}(j)+(S_i-S_j)^2+M\,\Bigr\}
		\]
		Which terms couple $i$ and $j$?
		What does that suggest about the inner minimisation?
	}
\end{frame}

\begin{frame}{Rewriting the recurrence}
	\[
		\mathrm{dp}(i)=S_i^2+M+\min_{j<i}\Bigl\{\underbrace{(-2S_j)}_{m_j}\cdot S_i+\underbrace{\mathrm{dp}(j)+S_j^2}_{b_j}\Bigr\}.
	\]
	Only the cross term $-2S_iS_j$ couples $i$ and $j$.
\end{frame}

\begin{frame}{A minimum over lines}
	Define the line $\ell_j(x)=m_jx+b_j$. Then
	\[
		\boxed{\;\mathrm{dp}(i)=S_i^2+M+\min_{j<i}\ell_j(S_i)\;}
	\]
	The inner term is the \alert{lower envelope} of $\ell_0,\dots,\ell_{i-1}$ evaluated at $S_i$.
	Each line $\ell_i$ is inserted online, once $\mathrm{dp}(i)$ is known.
\end{frame}

\begin{frame}{The monotone convex hull trick}
	Keep the envelope as a deque $Q$ of lines in insertion order; queries $S_i$ are non-decreasing.

	\begin{enumerate}
		\item \textbf{Query.} While $|Q|\ge2$ and $\ell_{Q_2}(S_i)\le\ell_{Q_1}(S_i)$, pop the front.
		Then $\mathrm{dp}(i)=S_i^2+M+\ell_{Q_1}(S_i)$.
		\item \textbf{Insert} $\ell_i$. While $|Q|\ge2$ and the last line is redundant, pop the back.
	\end{enumerate}

	\seprule

	Each line is pushed once and popped at most once: \alert{amortised $O(1)$ per state, $\Theta(n)$ in total}.
\end{frame}

\begin{frame}{The redundancy test}
	Last line $\ell_2$, its predecessor $\ell_1$, new line $\ell_i$; slopes $m_1>m_2>m_i$.
	\[
		\ell_2\text{ redundant}\iff (b_i-b_1)(m_1-m_2)\le(b_2-b_1)(m_1-m_i),
	\]
	that is, $\ell_1$ and $\ell_i$ intersect left of $\ell_1$ and $\ell_2$.
\end{frame}

\begin{frame}{When the data are not monotone}
	\audience[Choose the data structure]{%
		\begin{enumerate}
			\item[(a)] Slopes inserted in decreasing order, \emph{arbitrary} query points.
			\item[(b)] \emph{Arbitrary} slopes (e.g.\ some $c_l<0$), arbitrary queries, online.
		\end{enumerate}
	}
\end{frame}

\begin{frame}{When the data are not monotone: answers}
	\begin{enumerate}
		\item[(a)] Keep the hull as an array and \alert{binary-search} the query: $O(n\log n)$ total.
		\item[(b)] A \alert{Li Chao tree} over the query domain (or a dynamic hull): $O(\log C)$ per insertion and query,
		where $C$ is the size of the query domain.
	\end{enumerate}
\end{frame}

\begin{frame}{What structure was exploited?}
	The trick applies whenever the transition separates as
	\[
		\mathrm{dp}(i)=\alpha_i+\min_{j<i}\{\beta_j+\gamma_i\,\delta_j\}.
	\]
	Other structure in $\mathrm{cost}(j,i)$ gives other accelerations.
\end{frame}

% ---------------------------------------------------------------- Lever II
\begin{frame}{Lever II: domino tilings}
	Count the tilings of an $n\times m$ board (some cells may be blocked) by $1\times2$ dominoes,
	with $m$ small ($\le 16$) and $n$ large.

	\vspace{0.5em}
	\textbf{Row-by-row state.} $G_r(M')$ counts partial tilings in which rows $<r$ are fully covered and
	$M'\subseteq\{0,\dots,m-1\}$ is exactly the set of cells of row $r$ covered by vertical dominoes from row $r-1$.
\end{frame}

\begin{frame}{Row-by-row recurrence}
	\[
		G_0(\emptyset)=1,\qquad
		G_{r+1}(M')=\sum_{M\;\text{compatible with}\;M'}G_r(M),\qquad\text{answer }G_n(\emptyset).
	\]
	$M$ and $M'$ are compatible iff $M\cap M'=\emptyset$ and the cells of the row outside $M\cup M'$ split into
	runs of even length.

	\seprule

	$2^m$ states per row. Compatible pairs: up to $3^m$. Time $\Theta(n\,3^m)$.
\end{frame}

\begin{frame}{The edge count is the bottleneck}
	\audience[Fewer edges, same counts]{%
		There are only $2^m$ states per row, but $3^m$ edges between consecutive rows.
		Can the \textbf{same} counts be computed with fewer edges?
	}
\end{frame}

\begin{frame}{One cell at a time}
	\begin{itemize}
		\item Process cells in row-major order.
		\item State: the \alert{profile} of the next $m$ cells (covered or not).
		\item Each cell has $O(1)$ choices: skip, vertical domino, horizontal domino.
	\end{itemize}

	\seprule

	Same sums, regrouped. Edges per state drop from $2^m$ to $O(1)$.
\end{frame}

\begin{frame}{Broken profile DP}
	Number cells $t=rm+c$ in row-major order. $F_t(M)$, $M\in\{0,1\}^m$, counts partial tilings in which cells $<t$ are
	covered; bit $k$ of $M$ says whether cell $t+k$ is already covered.
	\[
		F_0(0)=1,\qquad \text{answer }F_{nm}(0).
	\]
\end{frame}

\begin{frame}{Broken profile DP: transitions}
	From $F_t(M)$, with cell $t=(r,c)$:
	\begin{align*}
		M_0=1:\;& F_{t+1}\bigl(M\gg1\bigr)\mathrel{+}=F_t(M)\\
		M_0=0:\;& F_{t+1}\bigl((M\gg1)\,|\,2^{m-1}\bigr)\mathrel{+}=F_t(M) && \text{vertical, if } r+1<n\\
		M_0=M_1=0:\;& F_{t+1}\bigl((M\gg1)\,|\,1\bigr)\mathrel{+}=F_t(M) && \text{horizontal, if } c+1<m
	\end{align*}

	\seprule

	States: $nm\,2^m$. In-edges: $\le 3$. Time $\Theta(nm\,2^m)$ versus $\Theta(n\,3^m)$.
\end{frame}
