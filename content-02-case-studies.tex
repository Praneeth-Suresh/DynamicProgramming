\section{Is It Dynamic Programming? Case Studies}

\begin{frame}{How this part works}
	For each problem, decide for every candidate solution:
	\begin{enumerate}
		\item Is it \textbf{correct}?
		\item Is it \textbf{the right tool}?
	\end{enumerate}
\end{frame}

\begin{frame}{Triage test for dynamic programming}
	\begin{enumerate}
		\item Is there a \emph{sufficient} state of polynomial size? (Markov property)
		\item Does the objective decompose over it? (optimal substructure)
		\item Is there no cheaper algorithm that exploits more structure?
	\end{enumerate}

	\seprule

	(1) and (2) make DP \emph{correct}; (3) decides whether it is \emph{good}.
\end{frame}

% ---------------------------------------------------------------- Case 1
\begin{frame}{Case 1: Split Array Largest Sum (LeetCode 410)}
	Given $a_1,\dots,a_n\ge 0$ and $k\le n$, split the array into $k$ non-empty
	contiguous parts so as to \alert{minimise the largest part sum}. Let $S_i=\sum_{l\le i}a_l$.

	\seprule

	\textbf{Example.} $a=(7,2,5,10,8)$, $k=2$: the parts $(7,2,5)\,|\,(10,8)$ give $\max\{14,18\}=18$, which is optimal.
\end{frame}

\begin{frame}{Case 1: candidate solutions}
	\begin{columns}[T,onlytextwidth]
		\column{0.52\textwidth}
		\begin{block}{Proposal A (recurrence)}
			\small
			Let $f(i,j)$ be the best value for the first $i$ elements in $j$ parts.
			\begin{align*}
				f(0,0)&=0, \quad f(i,0)=f(0,j)=\infty\\
				f(i,j)&=\min_{j-1\le p<i}\max\bigl\{f(p,j-1),\;S_i-S_p\bigr\}
			\end{align*}
			Answer $f(n,k)$.
		\end{block}

		\column{0.44\textwidth}
		\begin{block}{Proposal B (greedy)}
			\small
			Let $\bar S=\lceil S_n/k\rceil$. Sweep left to right and close the current part
			as soon as adding the next element would exceed $\bar S$ (until $k-1$ parts are closed).
		\end{block}
	\end{columns}
\end{frame}

\begin{frame}{Case 1: your verdict}
	\audience[Correct? Efficient?]{%
		For each proposal: \textbf{correct} or \textbf{incorrect}?
		If correct, is it \textbf{efficient}?
	}
\end{frame}

\begin{frame}{Case 1: Proposal A is correct, but slow}
	The state $(i,j)$ is sufficient: only the prefix length and the number of parts matter,
	and $(\max,\min)$ composes.
	\[
		\underbrace{nk}_{\text{states}}\times\underbrace{O(n)}_{\text{per state}}=\Theta(kn^2).
	\]
\end{frame}

\begin{frame}{Case 1: Proposal B is wrong}
	Take $a=(1,3,2)$, $k=2$, so $\bar S=3$.
	\begin{itemize}
		\item B closes $(1)$ and returns $(1)\,|\,(3,2)$: value $5$.
		\item $(1,3)\,|\,(2)$ achieves $4$.
	\end{itemize}
\end{frame}

\begin{frame}{Case 1: decide, then optimise}
	For a cap $L$, let $c(L)$ be the minimum number of parts with every sum $\le L$.
	\begin{itemize}
		\item Greedy ``extend each part as far as possible'' computes $c(L)$ in $O(n)$.
		\item $c$ is non-increasing in $L$, and a partition into fewer than $k$ parts can be refined.
		\item Answer: $\min\{L\in[\max_i a_i,\,S_n]\;:\;c(L)\le k\}$, by binary search.
	\end{itemize}
\end{frame}

\begin{frame}{Case 1: comparison}
	\begin{center}
		\begin{tabular}{@{}lc@{}}
			\toprule
			Method & Time\\
			\midrule
			Proposal A & $\Theta(kn^2)$\\
			A with monotone optimum (binary search over $p$) & $O(kn\log n)$\\
			Binary search on the answer + greedy & $O(n\log S_n)$\\
			\bottomrule
		\end{tabular}
	\end{center}
\end{frame}

\begin{frame}{Case 1: why the maximum helps}
	\audience[Minimising the maximum]{%
		\begin{enumerate}
			\item Which property of ``minimise the \emph{maximum}'' made the decision version
			greedy-checkable and monotone in $L$?
			\item Suppose we instead minimise the \textbf{sum of squares} of part sums. Does binary search
			on the answer still work? What would you use?
		\end{enumerate}
	}
\end{frame}

\begin{frame}{Case 1: answers}
	\begin{enumerate}
		\item The maximum commutes with refinement: splitting a part never increases the objective,
		and feasibility under a cap is local.
		\item No: there is no threshold structure. This is a genuine DP,
		\[
			f(i,j)=\min_p\; f(p,j-1)+(S_i-S_p)^2,
		\]
		and its transition has the form that the \alert{convex hull trick} exploits in Part~3.
	\end{enumerate}
\end{frame}

% ---------------------------------------------------------------- Case 2
\begin{frame}{Case 2: Maximum weight independent set on a tree}
	LeetCode 337 (\emph{House Robber III}); the path case is LeetCode 198.

	\vspace{0.5em}
	Tree $T=(V,E)$ rooted at $r$, weights $w_v\ge0$. Choose an independent set $I$ maximising $\sum_{v\in I}w_v$.

	\vspace{0.5em}
	Notation: $\ch(v)$ are the children and $\gc(v)$ the grandchildren of $v$.
\end{frame}

\begin{frame}{Case 2: candidate solutions}
	\begin{description}
		\item[\alert{Proposal A}] Greedy: repeatedly take the heaviest remaining vertex and delete its neighbours.
		\item[\alert{Proposal B}] $f(v)=\max\bigl\{\,w_v+\sum_{c\in\ch(v)}f(c),\;\sum_{c\in\ch(v)}f(c)\bigr\}$
		\item[\alert{Proposal C}] $f(v)=\max\bigl\{\,w_v+\sum_{g\in\gc(v)}f(g),\;\sum_{c\in\ch(v)}f(c)\bigr\}$
		\item[\alert{Proposal D}] $f(v,1)=w_v+\sum_{c}f(c,0)$ and $f(v,0)=\sum_c\max\{f(c,0),f(c,1)\}$; answer $\max_b f(r,b)$.
	\end{description}
\end{frame}

\begin{frame}{Case 2: your verdict}
	\audience[Which proposals are correct?]{%
		Find a counterexample for each proposal you reject.
	}
\end{frame}

\begin{frame}{Case 2: A and B are wrong}
	Path $2-3-2$:
	\begin{itemize}
		\item \textbf{A} takes the centre: value $3$. The optimum is $4$ (both ends).
		\item \textbf{B}, rooted at the centre: $f(\text{leaf})=2$, so $f(\text{root})=\max\{3+4,4\}=7$, which is infeasible.
		The term $w_v+\sum_c f(c)$ forgets that $f(c)$ may itself use $c$.
	\end{itemize}
\end{frame}

\begin{frame}{Case 2: C and D are correct}
	In D the state records whether $v$ is chosen:
	\begin{align*}
		f(v,1)&=w_v+\sum_{c}f(c,0)\\
		f(v,0)&=\sum_{c}\max_b f(c,b)
	\end{align*}
	C is D with the state projected out: $f(v)=\max_b f(v,b)$.

	\seprule

	Time $\Theta(\sum_v\deg v)=\Theta(n)$. D has $2n$ states and \emph{local} transitions.
\end{frame}

\begin{frame}{Case 2: what generalises}
	\begin{theorem}[folklore]
		Given a tree decomposition of width $\omega$ of an $n$-vertex graph, a maximum weight independent
		set can be computed in $O(2^{\omega}\,\omega\, n)$ time.
	\end{theorem}
	The state of a bag is the independent subset chosen inside it ($\le 2^{\omega+1}$ states).

	\seprule

	Trees ($\omega=1$), cycles ($\omega=2$), series-parallel graphs ($\omega\le2$), $k\times n$ grids ($\omega=k$).
\end{frame}

\begin{frame}{Case 2: from trees to cycles}
	\audience[Independent set on a cycle]{%
		Now let the graph be a \textbf{cycle} $C_n$ with weights $w_1,\dots,w_n$.
		The tree recurrence is no longer valid. Fix it, and give the complexity.
	}
\end{frame}

\begin{frame}{Case 2: the cycle}
	\begin{itemize}
		\item Fix the status $b_1$ of vertex $1$ and run the path DP on $1,\dots,n$ with that boundary condition
		(for $b_1=1$, vertex $n$ is forbidden). Two path DPs: $\Theta(n)$.
		\item Equivalently, enlarge the state to $(b_1,b_{\text{current}})$: $4$ states per position.
	\end{itemize}
\end{frame}

% ---------------------------------------------------------------- Case 3
\begin{frame}{Case 3: Minimum Initial Energy to Finish Tasks (LeetCode 1665)}
	Tasks $i=1,\dots,n$ with $(a_i,m_i)$, $a_i\le m_i$: to \emph{start} task $i$ you need at least $m_i$ energy,
	and finishing it consumes $a_i$. Choose the order $\pi$ to minimise the initial energy $E$.

	\seprule

	For an order $\pi$:
	\[
		E(\pi)=\max_{t}\Bigl(m_{\pi(t)}+\sum_{s<t}a_{\pi(s)}\Bigr),\qquad
		E^\star=\min_{\pi}E(\pi).
	\]
\end{frame}

\begin{frame}{Case 3: candidate solutions}
	Let $A(S)=\sum_{i\in S}a_i$.
	\begin{description}
		\item[\alert{Proposal A}] Sort by $m_i$ in \emph{decreasing} order.
		\item[\alert{Proposal B}] Sort by $a_i$ in \emph{increasing} order.
		\item[\alert{Proposal C}] $F(\emptyset)=0$,\\
		$F(S)=\min_{i\in S}\max\bigl\{F(S\setminus\{i\}),\;A(S\setminus\{i\})+m_i\bigr\}$;\\
		answer $F([n])$.
	\end{description}
\end{frame}

\begin{frame}{Case 3: your verdict}
	\audience[Correct? Useful?]{%
		\begin{enumerate}
			\item Which proposals are correct?
			\item Is C a ``dynamic programming solution''? Is it a \emph{useful} one?
		\end{enumerate}
	}
\end{frame}

\begin{frame}{Case 3: A and B are wrong}
	Tasks written $[a_i,m_i]$.
	\begin{itemize}
		\item \textbf{A:} $[1,2],[3,3]$. A runs $[3,3]$ first and needs $\max\{3,\,3+2\}=5$;
		running $[1,2]$ first needs $\max\{2,\,1+3\}=4$.
		\item \textbf{B:} $[1,1],[2,3]$. B needs $\max\{1,1+3\}=4$; the other order needs $\max\{3,2+1\}=3$.
	\end{itemize}
\end{frame}

\begin{frame}{Case 3: C is correct, but does not compress}
	The energy left after finishing a set $S$ is $E-A(S)$ \emph{whatever the order}, so
	the set is a sufficient state.

	\seprule

	But the state is the \alert{whole subset}: $2^n$ states, $\Theta(2^nn)$ time.
	The future depends on \emph{which} tasks remain, not on a number summarising them.
\end{frame}

\begin{frame}{Case 3: an exchange argument}
	Key $\kappa_i=m_i-a_i$. \alert{Claim:} an order sorted by decreasing $\kappa$ is optimal.

	\vspace{0.5em}
	Run $i$ immediately before $j$ with $\kappa_i\ge\kappa_j$:
	\begin{align*}
		E(i,j)&=\max\{a_i+\kappa_i,\;a_i+a_j+\kappa_j\}\\
		&\le a_i+a_j+\kappa_i\le \max\{m_j,\;a_j+m_i\}=E(j,i).
	\end{align*}
	Swapping adjacent out-of-order pairs never increases the requirement. Time $O(n\log n)$.
\end{frame}

% ---------------------------------------------------------------- Rapid fire
\begin{frame}{Rapid fire: DP, or not?}
	\audience[DP, something better, or no compact DP?]{%
		\begin{enumerate}
			\item Minimum number of coins summing to $V$.
			\item Unweighted interval scheduling (maximum number of disjoint intervals).
			\item Longest \emph{simple} path in a general directed graph.
			\item $0/1$ knapsack with $n=40$ items and weights up to $10^{12}$.
			\item Longest increasing subsequence.
			\item Shortest paths in a graph with cycles and non-negative weights.
		\end{enumerate}
	}
\end{frame}

\begin{frame}{Rapid fire: answers}
	\begin{enumerate}
		\item \textbf{DP.} State $=$ remaining amount.
		\item \textbf{Something better.} Earliest-finish-time greedy, $O(n\log n)$.
		\item \textbf{No compact DP.} NP-hard; subset DP is $\Theta(2^nn)$.
		\item \textbf{Something better.} Meet-in-the-middle, $O(2^{n/2}\,n)$.
		\item \textbf{DP, then better.} $O(n^2)$ DP; state \emph{(length $\to$ smallest tail)} gives $O(n\log n)$.
		\item \textbf{DP in disguise.} Cyclic Bellman equations; Dijkstra chooses the order (Part~4).
	\end{enumerate}
\end{frame}
