\begin{frame}{A motivating problem: minimum edit script}
	\textbf{Version control.} Given two versions of a file as token sequences
	$x \in \Sigma^m$ and $y \in \Sigma^n$, produce a cheapest sequence of edits that
	turns $x$ into $y$.

	\begin{itemize}
		\item \textbf{delete} a token of $x$ \hfill cost $1$
		\item \textbf{insert} a token of $y$ \hfill cost $1$
		\item \textbf{substitute} a token \hfill cost $1$ (cost $0$ if the tokens already agree)
	\end{itemize}

	\seprule

	Goal: compute $\mathrm{ED}(x,y)$, the minimum total cost, and an optimal script.
	A script is a sequence of $\le m+n$ local decisions: the number of scripts is
	exponential. What is a \alert{good} way to organise the search?
\end{frame}

\begin{frame}{State spaces}
	\small
	\textbf{Definition.} A \alert{state space} is a weighted digraph $G=(V,E,\cost)$ with source $s$ and target $t$:
	solutions of the problem correspond to $s\to t$ paths, and the objective is the sum of edge weights.

	\begin{itemize}
		\item A state $v$ summarises a \emph{partial solution}.
		\item \alert{Sufficiency (Markov property):} the feasible completions of $v$, and
		their costs, depend only on $v$, not on the path that reached it.
	\end{itemize}

	\seprule

	\textbf{Edit distance.} State $(i,j)$: ``$x_{1..i}$ has been transformed into $y_{1..j}$''. Source $(0,0)$, target $(m,n)$.
	\[
		\begin{array}{lll}
			(i-1,j)\to(i,j) & \text{delete } x_i & \cost=1\\
			(i,j-1)\to(i,j) & \text{insert } y_j & \cost=1\\
			(i-1,j-1)\to(i,j) & \text{match / substitute} & \cost=\indic{x_i\neq y_j}
		\end{array}
	\]
\end{frame}

\begin{frame}{Paths versus states}
	The number of $s\to t$ paths in the $n\times n$ grid with the three step types is the
	central Delannoy number $D(n,n)\sim c\,(3+2\sqrt2)^n$, while there are only $(n+1)^2$ states.

	\audience{Poll}{%
		For $m=n=20$: roughly how many edit scripts (paths) are there?\\[0.3em]
		(a)~$10^{6}$ \quad (b)~$10^{9}$ \quad (c)~$10^{12}$ \quad (d)~$>10^{14}$
	}

	\uncover<2->{%
		\begin{center}
			$D(20,20) = 260{,}543{,}813{,}797{,}441 \approx 2.6\cdot 10^{14}$ paths
			\quad versus \quad $441$ states.
		\end{center}
		Dynamic programming exists because \alert{many paths share the same sub-paths}:
		the state graph is a compressed representation of the solution space.%
	}
\end{frame}

\begin{frame}{Dynamic programming on a DAG}
	\small
	\textbf{Theorem.} Let $G=(V,E,\cost)$ be a DAG with source $s$, and $d(v)$ the minimum weight of an
	$s\to v$ path ($\infty$ if none). Then
	\[
		d(s)=0, \qquad d(v)=\min_{(u,v)\in E}\bigl\{\,d(u)+\cost(u,v)\,\bigr\}\quad (v\neq s).
	\]
	\textbf{Proof sketch.} An $s\to v$ path is an $s\to u$ path followed by an edge $(u,v)$ and its weight is the
	sum of the two parts. The choice of prefix does not constrain the last edge, so minimising each prefix is lossless
	(\alert{optimal substructure}). Acyclicity makes the equations well-founded: induct along a topological order.

	\seprule

	Cyclic graphs satisfy the same equations, but they no longer \emph{define} $d$ by induction (Part~4).
\end{frame}

\begin{frame}{A strategy for dynamic programming problems}
	\small
	\begin{enumerate}
		\item \textbf{Decompose.} Choose a state space: what must be remembered about a
		partial solution so that the future is independent of the past?
		\item \textbf{Recurrence.} State the \alert{base case} and the \alert{inductive step}
		expressing each state through states that precede it.
		\item \textbf{Evaluate bottom-up} in a topological order of the state DAG.
		\item \textbf{Reconstruct} (if the solution, not only its value, is needed) by
		recording or recomputing the minimising predecessor of each state.
	\end{enumerate}

	\seprule

	\textbf{Complexity is read off the DAG:}
	\[
		T \;=\; \sum_{v\in V} \bigl(\text{work to evaluate state } v\bigr) \;=\; |V|\times(\text{cost per state}) \;\approx\; |E|.
	\]
\end{frame}

\begin{frame}{Steps 1--2: the recurrence for edit distance}
	\small
	Let $f(i,j)$ be the minimum cost of transforming $x_{1..i}$ into $y_{1..j}$.

	\begin{align*}
		\text{base:}\quad & f(i,0)=i, \qquad f(0,j)=j\\[0.3em]
		\text{step:}\quad & f(i,j)=\min\begin{cases}
			f(i-1,j)+1 & \text{delete } x_i\\
			f(i,j-1)+1 & \text{insert } y_j\\
			f(i-1,j-1)+\indic{x_i\neq y_j} & \text{match / substitute}
		\end{cases}
	\end{align*}
	Answer: $\mathrm{ED}(x,y)=f(m,n)$.

	\seprule

	\textbf{Why correct?} Consider the \emph{last} operation of an optimal script for
	$(x_{1..i},y_{1..j})$. It is one of three kinds, and the script before it is a script
	for the corresponding smaller pair; it must itself be optimal, otherwise it could be replaced
	by a cheaper one.
\end{frame}

\begin{frame}{Step 4: reconstruction}
	\small
	Store (or recompute) the minimising predecessor of each state; walk back from $(m,n)$:
	\[
		(4,4)\xleftarrow{\text{insert n}}(4,3)\xleftarrow{\text{match w}}(3,2)\xleftarrow{\text{match a}}(2,1)\xleftarrow{\text{match l}}(1,0)\xleftarrow{\text{delete f}}(0,0).
	\]
	\begin{center}
		\ttfamily
		\begin{tabular}{@{}cccccc@{}}
			f & l & a & w & - \\
			- & l & a & w & n \\
			\textrm{del} & = & = & = & \textrm{ins}
		\end{tabular}
	\end{center}

	\seprule

	\begin{itemize}
		\item Recomputing the argmin on the way back needs only the value table.
		\item The value table costs $\Theta(mn)$ memory. Rolling two rows gives
		$\Theta(\min(m,n))$ but loses reconstruction; \alert{Hirschberg's divide-and-conquer}
		restores it at twice the time and linear space.
	\end{itemize}
\end{frame}

\begin{frame}{Complexity and its limits}
	$|V|=(m+1)(n+1)$, at most $3$ predecessors each: $\Theta(mn)$ time.

	\begin{itemize}
		\item \textbf{Output-sensitive.} If $\mathrm{ED}\le d$, only the band $|i-j|\le d$ matters:
		$O(d\cdot\min(m,n))$ (Ukkonen); Myers' $O(nd)$ diff is the engine behind \texttt{git diff}.
		\item \textbf{Lower bound.} No $O(n^{2-\varepsilon})$ algorithm exists unless SETH
		fails (Backurs--Indyk, STOC 2015).
	\end{itemize}

	\seprule

	Lesson: \emph{the state space fixes a baseline}; structure of the instance or of
	the cost function can beat it, and conditional lower bounds tell us when to stop trying.
\end{frame}

\begin{frame}{Is the state space really sufficient?}
	\small
	\audience{Discuss}{%
		Allow \textbf{transpositions} of adjacent tokens ($ab\to ba$) at cost $1$.
		(1)~Which edge do you add to the DAG, and when? (2)~Does the complexity change?
		(3)~Is the result exactly the Damerau--Levenshtein distance?
	}

	\uncover<2->{%
		\begin{enumerate}
			\item Add $(i-2,j-2)\to(i,j)$ with cost $1$ when $x_i=y_{j-1}$ and $x_{i-1}=y_j$.
			\item Still $\Theta(mn)$: one more $O(1)$ in-edge.
			\item \alert{No.} This is the \emph{optimal string alignment} distance (no substring is edited twice).
			For $x=\texttt{CA}$, $y=\texttt{ABC}$ it gives $3$, the true distance is $2$ (\texttt{CA}$\to$\texttt{AC}$\to$\texttt{ABC}).
			Our edges are \alert{not sufficient}: the exact algorithm needs edges reaching back to the last occurrence of a symbol (see backup).
		\end{enumerate}
	}
\end{frame}

\begin{frame}{Part 1 in one slide}
	What made this a dynamic program?
	\begin{enumerate}
		\item \textbf{A sufficient state}, giving a graph that compresses exponentially many solutions.
		\item \textbf{Acyclicity}, giving a well-founded order of evaluation.
		\item \textbf{Additive (or otherwise decomposable) objective}, giving optimal substructure.
	\end{enumerate}

	\seprule

	\textbf{Design is in step 1.} Steps 2--4 are mechanical once the state space is right,
	and so is the complexity: $|V|\times$ cost per state.

\end{frame}
